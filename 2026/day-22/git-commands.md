# Set your Git username globally
git config --global user.name "Your Name"

# Set your Git email address globally (should match your GitHub account)
git config --global user.email "your.email@example.com"

# Verify your configured username
git config user.name

# Verify your configured email address
git config user.email

# To get the status 

git status

# To add a file into stating 

git add <file_name>
git add .   #add all file 

# To commit a file 

git commit -m "Commit message"

# To see log 

git log --oneline
